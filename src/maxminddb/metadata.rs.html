13/// Metadata about the MaxMind DB file.
14#[derive(Deserialize, Serialize, Clone, Debug, PartialEq, Eq)]
15pub struct Metadata {
16    /// Major version of the binary format (always 2).
17    pub binary_format_major_version: u16,
18    /// Minor version of the binary format.
19    /// Minor versions are accepted for forward compatibility.
20    pub binary_format_minor_version: u16,
21    /// Unix timestamp when the database was built.
22    pub build_epoch: u64,
23    /// Database type (e.g., "GeoIP2-City", "GeoLite2-Country").
24    pub database_type: String,
25    /// Map of language codes to database descriptions.
26    #[serde(deserialize_with = "deserialize_description")]
27    pub description: BTreeMap<String, String>,
28    /// IP version supported (4 or 6).
29    pub ip_version: u16,
30    /// Languages available in the database.
31    pub languages: Vec<String>,
32    /// Number of nodes in the search tree.
33    pub node_count: u32,
34    /// Size of each record in bits (24, 28, or 32).
35    pub record_size: u16,
36}

38/// Deserializes the `description` map, rejecting repeated language keys.
39///
40/// The spec requires metadata keys to be unique. Serde's derived struct
41/// deserializer already rejects repeated top-level keys, but a plain
42/// `BTreeMap` would silently keep the last value for a repeated key in the
43/// nested `description` map.
44fn deserialize_description<'de, D>(deserializer: D) -> Result<BTreeMap<String, String>, D::Error>
45where
46    D: Deserializer<'de>,
47{
48    struct DescriptionVisitor;
49
50    impl<'de> Visitor<'de> for DescriptionVisitor {
51        type Value = BTreeMap<String, String>;
52
53        fn expecting(&self, formatter: &mut fmt::Formatter) -> fmt::Result {
54            formatter.write_str("a map of language codes to descriptions")
55        }
56
57        fn visit_map<A>(self, mut map: A) -> Result<Self::Value, A::Error>
58        where
59            A: MapAccess<'de>,
60        {
61            let mut description = BTreeMap::new();
62            while let Some((language, text)) = map.next_entry::<String, String>()? {
63                match description.entry(language) {
64                    Entry::Vacant(entry) => {
65                        entry.insert(text);
66                    }
67                    Entry::Occupied(entry) => {
68                        return Err(de::Error::custom(format!(
69                            "duplicate key `{}` in metadata field `description`",
70                            entry.key()
71                        )));
72                    }
73                }
74            }
75            Ok(description)
76        }
77    }
78
79    deserializer.deserialize_map(DescriptionVisitor)
80}

82impl Metadata {
83    /// Returns the database build time as a `SystemTime`.
84    ///
85    /// This converts the `build_epoch` Unix timestamp to a `SystemTime`.
86    /// If `build_epoch` is too large to represent on this platform, this
87    /// returns an [`InvalidDatabase`](MaxMindDbError::InvalidDatabase) error.
88    ///
89    /// # Example
90    ///
91    /// ```
92    /// use maxminddb::Reader;
93    ///
94    /// let reader = Reader::open_readfile("test-data/test-data/GeoIP2-City-Test.mmdb").unwrap();
95    /// let build_time = reader.metadata().build_time().unwrap();
96    /// println!("Database built: {:?}", build_time);
97    /// ```
98    #[inline]
99    pub fn build_time(&self) -> Result<SystemTime, MaxMindDbError> {
100        UNIX_EPOCH
101            .checked_add(Duration::from_secs(self.build_epoch))
102            .ok_or_else(|| {
103                MaxMindDbError::invalid_database(format!(
104                    "build_epoch - Unix timestamp is too large to represent as SystemTime: {}",
105                    self.build_epoch
106                ))
107            })
108    }
109}
